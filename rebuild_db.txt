-- ============================================================
-- 股票練習系統 DB 重建腳本
-- 資料庫：stock_practice_db
-- 版本：v1.3（2026-06-04）
-- 使用方式：在 TiDB Cloud SQL Editor 全選貼上執行
-- ============================================================


-- ============================================================
-- STEP 1: 砍掉所有舊表
-- ============================================================
USE stock_practice_db;
SET FOREIGN_KEY_CHECKS = 0;
DROP TABLE IF EXISTS stock_character_notes;
DROP TABLE IF EXISTS trading_practice_records;
DROP TABLE IF EXISTS daily_newsletter;
DROP TABLE IF EXISTS daily_strategy_results;
DROP TABLE IF EXISTS stock_eps;
DROP TABLE IF EXISTS stock_daily_data;
DROP TABLE IF EXISTS stock_info;
SET FOREIGN_KEY_CHECKS = 1;


-- ============================================================
-- STEP 2: 重建所有表
-- ============================================================

-- TABLE 1: 股票基本母檔表（必須最先建立）
CREATE TABLE stock_info (
    stock_code    VARCHAR(10)  NOT NULL COMMENT '股票代碼，如 2330.TW',
    stock_name    VARCHAR(50)  NOT NULL COMMENT '股票中文名稱',
    market_type   VARCHAR(20)  DEFAULT NULL COMMENT '上市 / 上櫃 / 美股',
    industry_type VARCHAR(50)  DEFAULT NULL COMMENT '產業別',
    created_at    TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (stock_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE 2: 每日K線與籌碼數據表
CREATE TABLE stock_daily_data (
    stock_code     VARCHAR(10)    NOT NULL COMMENT '股票代碼',
    trade_date     DATE           NOT NULL COMMENT '交易日期',
    open_price     DECIMAL(10, 2) NOT NULL COMMENT '開盤價',
    high_price     DECIMAL(10, 2) NOT NULL COMMENT '最高價',
    low_price      DECIMAL(10, 2) NOT NULL COMMENT '最低價',
    close_price    DECIMAL(10, 2) NOT NULL COMMENT '收盤價',
    volume         BIGINT         NOT NULL COMMENT '交易張數（÷1000換算）',
    foreign_buy    BIGINT DEFAULT 0 COMMENT '外資買賣超（正買負賣）',
    investment_buy BIGINT DEFAULT 0 COMMENT '投信買賣超',
    dealer_buy     BIGINT DEFAULT 0 COMMENT '自營商買賣超',
    PRIMARY KEY (stock_code, trade_date),
    KEY idx_trade_date (trade_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE 3: 每日策略選股結果表
CREATE TABLE daily_strategy_results (
    id               BIGINT AUTO_INCREMENT NOT NULL,
    trade_date       DATE           NOT NULL COMMENT '篩選日期',
    stock_code       VARCHAR(10)    NOT NULL COMMENT '股票代碼',
    is_chose_trigger TINYINT(1)     NOT NULL DEFAULT 0 COMMENT 'CHOSE型態掃描是否觸發',
    pattern_type     VARCHAR(50)    DEFAULT NULL COMMENT '技術型態，如：高窄旗型',
    chose_reason     VARCHAR(255)   DEFAULT NULL COMMENT '型態觸發原因',
    is_drive_trigger TINYINT(1)     NOT NULL DEFAULT 0 COMMENT 'DRIVE大戶動態是否觸發',
    drive_score      INT            DEFAULT NULL COMMENT '大戶動態評分，如：30, 50, 100',
    rs_rating        DECIMAL(5, 1)  DEFAULT NULL COMMENT 'RS相對強度指標',
    chip_feature     VARCHAR(100)   DEFAULT NULL COMMENT '吸籌特徵',
    suggested_entry  DECIMAL(10, 2) DEFAULT NULL COMMENT '推薦佈局價',
    stop_loss        DECIMAL(10, 2) DEFAULT NULL COMMENT '防禦停損價',
    target_price_1   DECIMAL(10, 2) DEFAULT NULL COMMENT '第一停利目標價',
    target_price_2   DECIMAL(10, 2) DEFAULT NULL COMMENT '第二停利目標價',
    backtest_win_rate   DECIMAL(5, 2) DEFAULT NULL COMMENT '策略3Y回測勝率%',
    backtest_total_pnl  DECIMAL(7, 2) DEFAULT NULL COMMENT '策略3Y回測總報酬%',
    PRIMARY KEY (id),
    UNIQUE KEY uq_date_code (trade_date, stock_code),
    KEY idx_date_modules (trade_date, is_chose_trigger, is_drive_trigger),
    CONSTRAINT fk_strategy_stock FOREIGN KEY (stock_code) REFERENCES stock_info (stock_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE 4: EPS季報數據表
CREATE TABLE stock_eps (
    id             BIGINT AUTO_INCREMENT NOT NULL,
    stock_code     VARCHAR(10)   NOT NULL COMMENT '股票代碼',
    fiscal_year    SMALLINT      NOT NULL COMMENT '會計年度，如 2025',
    fiscal_quarter TINYINT       NOT NULL COMMENT '季度 1~4',
    eps            DECIMAL(8, 2) DEFAULT NULL COMMENT '每股盈餘（元）',
    announced_date DATE          DEFAULT NULL COMMENT '財報公告日',
    created_at     TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_eps (stock_code, fiscal_year, fiscal_quarter),
    KEY idx_eps_stock (stock_code),
    CONSTRAINT fk_eps_stock FOREIGN KEY (stock_code) REFERENCES stock_info (stock_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE 5: 模擬交易與練習紀錄表
CREATE TABLE trading_practice_records (
    trade_id            BIGINT AUTO_INCREMENT NOT NULL,
    user_id             VARCHAR(50)    NOT NULL DEFAULT 'default_user' COMMENT '使用者識別 ID',
    stock_code          VARCHAR(10)    NOT NULL COMMENT '股票代碼',
    practice_mode       VARCHAR(20)    NOT NULL DEFAULT 'DAILY_DECISION' COMMENT '練習模式: BLIND_120 或 DAILY_DECISION',
    practice_start_date DATE           DEFAULT NULL COMMENT '練習啟動的歷史基準日期',
    entry_date          DATE           DEFAULT NULL COMMENT '模擬買進的歷史日期',
    entry_price         DECIMAL(10, 2) DEFAULT NULL COMMENT '模擬買進價格',
    entry_qty           INT            DEFAULT 1000 COMMENT '模擬交易數量',
    user_stop_loss      DECIMAL(10, 2) DEFAULT NULL COMMENT '停損價',
    user_take_profit    DECIMAL(10, 2) DEFAULT NULL COMMENT '停利價',
    exit_date           DATE           DEFAULT NULL COMMENT '實際出場日期',
    exit_price          DECIMAL(10, 2) DEFAULT NULL COMMENT '實際出場價格',
    status              VARCHAR(20)    NOT NULL DEFAULT '純觀望' COMMENT '狀態: 純觀望 / 持股中 / 已結案_停損 / 已結案_停利 / 已結案_主動出場',
    pnl_percent         DECIMAL(5, 2)  DEFAULT NULL COMMENT '損益百分比%',
    created_at          TIMESTAMP      DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (trade_id),
    KEY idx_practice_user (user_id, practice_mode),
    CONSTRAINT fk_practice_stock FOREIGN KEY (stock_code) REFERENCES stock_info (stock_code)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE 6: 個股股性標籤與覆盤筆記表
CREATE TABLE stock_character_notes (
    note_id        BIGINT AUTO_INCREMENT NOT NULL,
    user_id        VARCHAR(50)  NOT NULL DEFAULT 'default_user' COMMENT '使用者識別 ID',
    stock_code     VARCHAR(10)  NOT NULL COMMENT '股票代碼',
    trade_id       BIGINT       DEFAULT NULL COMMENT '關聯的交易練習ID',
    stock_tags     VARCHAR(255) DEFAULT NULL COMMENT '逗號分隔股性標籤，如: 假突破常客,投信鎖碼大戶股',
    review_content TEXT         COMMENT '個人覆盤心得',
    updated_at     TIMESTAMP    DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (note_id),
    KEY idx_note_stock (stock_code),
    KEY idx_note_user (user_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- TABLE 7: 每日選股信件表
CREATE TABLE daily_newsletter (
    id             BIGINT AUTO_INCREMENT NOT NULL,
    mail_date      DATE          NOT NULL COMMENT '信件對應的交易日期',
    source_tag     VARCHAR(50)   DEFAULT NULL COMMENT '信件來源標籤，如：CHOSE日報、DRIVE週報',
    subject        VARCHAR(255)  DEFAULT NULL COMMENT '信件主旨',
    content_text   LONGTEXT      DEFAULT NULL COMMENT '信件純文字內容',
    content_html   LONGTEXT      DEFAULT NULL COMMENT '信件HTML原文',
    is_read        TINYINT(1)    DEFAULT 0 COMMENT '已讀標記（0:未讀, 1:已讀）',
    created_at     TIMESTAMP     DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_mail_date_source (mail_date, source_tag),
    KEY idx_mail_date (mail_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- STEP 3: 驗證（執行後應看到 7 張表）
-- ============================================================
SHOW TABLES;
